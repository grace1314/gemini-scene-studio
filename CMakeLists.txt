cmake_minimum_required(VERSION 3.20)
project(GeminiSceneStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d features2d)
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/3rdparty/open3d")
find_package(Open3D REQUIRED)
add_library(studio_core libs/core.cpp)
target_include_directories(studio_core PUBLIC include 3rdparty/include)
target_link_libraries(studio_core PUBLIC OpenSSL::Crypto Threads::Threads)
add_executable(studio_server apps/server.cpp)
target_link_libraries(studio_server PRIVATE studio_core)
add_executable(studio_geometry apps/geometry.cpp)
add_executable(studio_public_demo apps/public_demo.cpp)
target_include_directories(studio_public_demo PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(studio_public_demo PRIVATE studio_core ${OpenCV_LIBS})
add_library(studio_reconstruction libs/reconstruction.cpp)
target_include_directories(studio_reconstruction PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(studio_reconstruction PUBLIC studio_core ${OpenCV_LIBS} Open3D::Open3D)
target_link_libraries(studio_geometry PRIVATE studio_reconstruction)
target_link_options(studio_geometry PRIVATE "-Wl,-rpath-link,${CMAKE_SOURCE_DIR}/3rdparty/open3d/lib" "-Wl,--disable-new-dtags")
set_target_properties(studio_geometry PROPERTIES BUILD_RPATH "${CMAKE_SOURCE_DIR}/3rdparty/open3d/lib")
enable_testing()
add_executable(studio_core_tests tests/core_test.cpp)
target_link_libraries(studio_core_tests PRIVATE studio_core)
target_compile_options(studio_core_tests PRIVATE -UNDEBUG)
add_test(NAME core COMMAND studio_core_tests)
